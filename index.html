2. Cálculo Diferencial — Tracción Tendinosa UHMWPE

Fuerza Tendinosa $T(t)$ y Derivada Temporal $\text{RFD}(t)$

RFD(t) = dT/dt
Función Inicial (Tensión Cable):
T(t) = T_max · (1 - e^(-α·t))
GeoGebra: T(x) = 20*(1 - exp(-5x))
Derivada (Rate of Force Dev.):
RFD(t) = dT/dt = α·T_max·e^(-α·t)
GeoGebra: RFD(x) = 100*exp(-5x)
Instante de tracción $t$: 0.25 s T(0.25) = 14.27 N | RFD = 28.65 N/s

Código Integrador C++ (Arduino IDE — Placa: ESP32S3 Dev Module)

Unifica el Esquema 1 (5 Piezos bidireccionales a 250 Hz) y el Esquema 2 (Movimiento Angular → Servo MG996R + Relé FES) y transmite telemetría compatible con este Dashboard. 

#include <ESP32Servo.h>

Servo motorPolea;

// =========================================================
// PINES DEL ESQUEMA 1: 5 DISCOS PIEZOELÉCTRICOS (12 MM)
// =========================================================
const int numPiezos = 5;
// Orden: Pulgar (5), Índice (1), Medio (2), Anular (3), Meñique (10)
const int pinesPiezos[numPiezos] = {5, 1, 2, 3, 10};
const char* nombresDedos[numPiezos] = {"Pulgar", "Indice", "Medio", "Anular", "Menique"};

// Umbral por encima del apoyo constante del dedo para detectar contacto externo (0 a 4095)
const int umbralContactoExterno = 1200; 
// Frecuencia de vibración háptica (250 Hz para corpúsculos de Pacini)
const int frecuenciaVibracion = 250; 
const int duracionVibracionMs = 200; // Duración del pulso vibratorio en milisegundos

// =========================================================
// PINES DEL ESQUEMA 2: GIROSCOPIO/ÁNGULO, SERVO Y RELÉ FES
// =========================================================
const int pinSensorAngular = 4; // Sensor de movimiento angular (GPIO 4)
const int pinReleFES       = 6; // Módulo Relé conectado al FES (GPIO 6)
const int pinMotorServo    = 7; // Servomotor MG996R (GPIO 7)

// Variables para detectar el movimiento angular
int anguloAnterior = -1;
const int umbralMovimientoAngular = 3; // Grados mínimos de cambio para considerar movimiento
unsigned long ultimoMovimientoMs = 0;
const unsigned long tiempoMantenerFES = 500; // Mantiene el FES activo mientras haya giro (ms)

// Función para hacer vibrar un piezo específico a 250 Hz y devolverlo a modo sensor
void emitirVibracionPiezo(int pin) {
  pinMode(pin, OUTPUT);
  // 250 Hz equivale a un periodo de 4000 microsegundos (2000 us HIGH, 2000 us LOW)
  int ciclos = (duracionVibracionMs * 1000L) / 4000L;
  for (int i = 0; i < ciclos; i++) {
    digitalWrite(pin, HIGH);
    delayMicroseconds(2000);
    digitalWrite(pin, LOW);
    delayMicroseconds(2000);
  }
  // Regresar el pin a entrada analógica para seguir leyendo contactos
  pinMode(pin, INPUT);
}

void setup() {
  Serial.begin(115200);

  // Configurar pines de los 5 piezos inicialmente como entradas
  for (int i = 0; i < numPiezos; i++) {
    pinMode(pinesPiezos[i], INPUT);
  }

  // Configurar Relé del FES apagado al inicio
  pinMode(pinReleFES, OUTPUT);
  digitalWrite(pinReleFES, LOW);

  // Configurar Servomotor MG996R a 50 Hz
  ESP32PWM::allocateTimer(0);
  motorPolea.setPeriodHertz(50);
  motorPolea.attach(pinMotorServo, 500, 2500);

  // Lectura inicial del ángulo
  int lecturaInicial = analogRead(pinSensorAngular);
  anguloAnterior = map(lecturaInicial, 0, 4095, 0, 180);
  motorPolea.write(anguloAnterior);

  Serial.println("=== SISTEMA INTEGRADOR INICIADO ===");
}

void loop() {
  // ---------------------------------------------------------
  // 1. ESQUEMA 1: LECTURA DE PIEZOS Y RESPUESTA DE VIBRACIÓN
  // ---------------------------------------------------------
  for (int i = 0; i < numPiezos; i++) {
    int lecturaPiezo = analogRead(pinesPiezos[i]);

    // Si supera el umbral de contacto externo (ignora apoyo constante)
    if (lecturaPiezo >= umbralContactoExterno) {
      Serial.print("¡Contacto externo en ");
      Serial.print(nombresDedos[i]);
      Serial.print(" (Valor: ");
      Serial.print(lecturaPiezo);
      Serial.println(")! -> Emitiendo vibracion haptica (250 Hz).");

      // Invierte la función del piezo para que vibre y estimule el dedo
      emitirVibracionPiezo(pinesPiezos[i]);
    }
  }

  // ---------------------------------------------------------
  // 2. ESQUEMA 2: MOVIMIENTO ANGULAR -> SERVO + FES
  // ---------------------------------------------------------
  int lecturaAngular = analogRead(pinSensorAngular);
  int anguloActual = map(lecturaAngular, 0, 4095, 0, 180);

  // Verificar si existe una variación angular real respecto a la posición previa
  if (abs(anguloActual - anguloAnterior) >= umbralMovimientoAngular) {
    Serial.print("Movimiento angular detectado -> Angulo: ");
    Serial.print(anguloActual);
    Serial.println("° | Girando Servo y Activando FES.");

    // Respuesta 1: El servomotor gira junto al movimiento angular
    motorPolea.write(anguloActual);

    // Respuesta 2: Se activa el relé para que el FES comience a funcionar
    digitalWrite(pinReleFES, HIGH);

    anguloAnterior = anguloActual;
    ultimoMovimientoMs = millis();
  }

  // Si el movimiento angular se detiene, se apaga el relé del FES tras medio segundo
  if (millis() - ultimoMovimientoMs > tiempoMantenerFES) {
    digitalWrite(pinReleFES, LOW);
  }

  delay(15); // Pausa breve de estabilidad del bucle
}
